// Dentro do seu arquivo...

const HyperbolicUniverse = ({ nodes, edges, manifold, playSonar, searchTerm, activeFilter }: any) => {
  const meshRef = useRef<THREE.InstancedMesh>(null);
  const [hovered, setHovered] = useState<NodeData | null>(null);
  const dummy = useMemo(() => new THREE.Object3D(), []);
  const color = useMemo(() => new THREE.Color(), []);

  // Isola os nós de Elite (Top 15 por energia) para ganhar Labels 3D
  const eliteNodes = useMemo(() => {
    return [...nodes].sort((a, b) => b.energy - a.energy).slice(0, 15);
  }, [nodes]);

  useFrame(() => {
    if (!meshRef.current) return;
    nodes.forEach((node: NodeData, i: number) => {
      meshRef.current!.getMatrixAt(i, dummy.matrix);
      const current = new THREE.Vector3().setFromMatrixPosition(dummy.matrix);
      const target = new THREE.Vector3(node.x, node.y, node.z);
      
      // SISTEMA DE FILTRO E BUSCA VISUAL (Dimming effect)
      const term = searchTerm.toLowerCase();
      const isSearchMatch = term === '' || node.id.includes(term) || node.node_type.toLowerCase().includes(term);
      const isFilterMatch = !activeFilter || node.node_type === activeFilter;
      const isHighlighted = isSearchMatch && isFilterMatch;

      // Animação de Movimento
      if (current.distanceTo(target) > 0.001) {
        current.lerp(target, 0.08);
      }
      
      // Tamanho e Posição
      const baseRadius = isHighlighted ? (0.01 + node.energy * 0.03) : 0.005; // Encolhe nós filtrados
      const radius = getVisualRadius(node, baseRadius);
      dummy.position.copy(current); 
      dummy.scale.set(radius, radius, 1);
      if (manifold === 'RIEMANN') dummy.lookAt(0,0,0);
      dummy.updateMatrix(); 
      meshRef.current!.setMatrixAt(i, dummy.matrix);

      // Cores Reativas (Cyberpunk)
      if (!isHighlighted) {
        color.setHex(0x0f172a); // Apaga completamente na escuridão
      } else {
        if (node.node_type === 'Pruned') color.setHex(0x334155);
        else if (node.node_type === 'Episodic') color.setHex(0x00f0ff);
        else color.setHex(0x00ff66);

        if (node.energy > 0.8) {
          color.setHex(0xff00ff);
          color.multiplyScalar(2.5); // Ativa o Bloom
        }
      }
      meshRef.current!.setColorAt(i, color);
    });
    
    meshRef.current.instanceMatrix.needsUpdate = true;
    if (meshRef.current.instanceColor) meshRef.current.instanceColor.needsUpdate = true;
  });

  return (
    <group>
      <HyperbolicParticles nodes={nodes} edges={edges} />
      
      {/* LABELS 3D DE ALTA PERFORMANCE (SDF) PARA OS ELITES */}
      {eliteNodes.map(node => (
        <Text 
          key={`label-${node.id}`} 
          position={[node.x, node.y - 0.03, node.z]} 
          fontSize={0.02} 
          color="#ffffff" 
          anchorX="center" 
          anchorY="middle"
          outlineWidth={0.002}
          outlineColor="#000000"
        >
          {node.node_type.toUpperCase()}
        </Text>
      ))}

      <instancedMesh 
        ref={meshRef} args={[undefined, undefined, nodes.length]}
        onPointerMove={(e) => {
          e.stopPropagation();
          if (e.instanceId !== undefined && nodes[e.instanceId] !== hovered) {
            const n = nodes[e.instanceId];
            setHovered(n);
            if (n.energy > 0.8) playSonar();
          }
        }}
        onPointerOut={() => setHovered(null)}
      >
        <circleGeometry args={[1, 32]} />
        <meshBasicMaterial toneMapped={false} />
      </instancedMesh>

      {/* Tooltip HTML mantido igual... */}
    </group>
  );
};
