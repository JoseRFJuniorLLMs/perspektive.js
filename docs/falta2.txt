### 1. Schrödinger Edges Shader (O Pulsar da Incerteza)

Para representar arestas que estão em superposição, não podemos usar uma linha sólida. Precisamos de um *fragment shader* que use ruído procedimental baseado na probabilidade de colapso da aresta.

**Caminho sugerido:** `src/edges/SchrodingerEdgeMaterial.ts`

```glsl
// Fragment Shader
uniform float time;
uniform float probability; // Vem do NietzscheDB (0.0 a 1.0)
varying vec2 vUv;

float noise(vec2 p) {
    return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453);
}

void main() {
    float pulse = sin(time * 5.0) * 0.5 + 0.5;
    // Quanto menor a probabilidade, mais "estática" e transparência
    float threshold = 1.0 - probability;
    float n = noise(vUv * time);
    
    if (n < threshold * pulse) {
        discard; // Cria o efeito de "flicker" quântico
    }
    
    vec3 color = vec3(0.0, 1.0, 1.0); // Cyan base
    gl_FragColor = vec4(color, probability * pulse);
}

```

### 1. O Shader de Incerteza (Schrödinger Edges)

Este material substitui as linhas estáticas. Ele faz a aresta "cintilar" (flicker) com ruído proporcional à probabilidade de colapso.

**Ficheiro:** `src/materials/SchrodingerEdgeMaterial.ts`

```typescript
import * as THREE from 'three';

const SchrodingerVertexShader = `
  varying vec2 vUv;
  void main() {
    vUv = uv;
    gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
  }
`;

const SchrodingerFragmentShader = `
  uniform float time;
  uniform float probability;
  uniform vec3 color;
  varying vec2 vUv;

  float noise(vec2 p) {
    return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453);
  }

  void main() {
    float edgeFactor = sin(vUv.x * 3.14159); // Suaviza as pontas
    float pulse = (sin(time * 10.0) * 0.5 + 0.5);
    float n = noise(vUv + time * 0.1);
    
    // Se a probabilidade for baixa, a linha "quebra" e falha
    if (n > probability + (pulse * 0.2)) {
      discard;
    }

    float alpha = probability * edgeFactor * (0.5 + pulse * 0.5);
    gl_FragColor = vec4(color, alpha);
  }
`;

export class SchrodingerEdgeMaterial extends THREE.ShaderMaterial {
  constructor(color = new THREE.Color('#00f0ff')) {
    super({
      uniforms: {
        time: { value: 0 },
        probability: { value: 0.5 },
        color: { value: color }
      },
      vertexShader: SchrodingerVertexShader,
      fragmentShader: SchrodingerFragmentShader,
      transparent: true,
      blending: THREE.AdditiveBlending
    });
  }
}

```

### 2. O Motor de Fluxo Cinético (Heat Flow Diffusion)

Este código injeta partículas nas geodésicas para mostrar o "pensamento" (calor) a viajar entre os nós conforme o algoritmo de Chebyshev no backend.

**Ficheiro:** `src/engine/KineticFlow.ts`

```typescript
import * as THREE from 'three';
import { calculateGeodesic } from '../math/poincare';

export class KineticFlowEngine {
  private particles: THREE.InstancedMesh;
  private particleCount: number = 5000;
  private dummy = new THREE.Object3D();

  constructor(scene: THREE.Scene) {
    const geo = new THREE.SphereGeometry(0.005, 8, 8);
    const mat = new THREE.MeshBasicMaterial({ color: '#ff00ff', transparent: true, opacity: 0.8 });
    this.particles = new THREE.InstancedMesh(geo, mat, this.particleCount);
    scene.add(this.particles);
  }

  update(activeEdges: any[], time: number) {
    let instanceIdx = 0;

    activeEdges.forEach(edge => {
      // Só desenha fluxo em arestas com "calor" (ativação)
      if (edge.activation > 0.1 && instanceIdx < this.particleCount) {
        const geodesic = calculateGeodesic(edge.source.pos, edge.target.pos);
        
        // Distribui 3 partículas por aresta ativa
        for (let i = 0; i < 3; i++) {
          const t = (time * 0.5 + (i / 3)) % 1.0;
          const pos = this.getPointOnGeodesic(geodesic, t);
          
          this.dummy.position.set(pos.x, pos.y, pos.z || 0);
          this.dummy.scale.setScalar(edge.activation);
          this.dummy.updateMatrix();
          this.particles.setMatrixAt(instanceIdx++, this.dummy.matrix);
        }
      }
    });
    this.particles.instanceMatrix.needsUpdate = true;
    this.particles.count = instanceIdx;
  }

  private getPointOnGeodesic(geo: any, t: number) {
    // Implementação da tua fórmula de arcos ortogonais
    const angle = geo.startAngle + (geo.endAngle - geo.startAngle) * t;
    return {
      x: geo.centerX + Math.cos(angle) * geo.radius,
      y: geo.centerY + Math.sin(angle) * geo.radius,
      z: 0
    };
  }
}

```

### 1. O Shader Quântico (Incerteza das Schrödinger Edges)

**Arquivo:** `src/materials/SchrodingerMaterial.ts`

```typescript
import * as THREE from 'three';

const SchrodingerGLSL = {
  vertex: `
    varying vec2 vUv;
    void main() {
      vUv = uv;
      gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
    }
  `,
  fragment: `
    uniform float time;
    uniform float probability; // 0.0 (incerto) a 1.0 (colapsado)
    uniform vec3 color;
    varying vec2 vUv;

    float hash(vec2 p) {
      return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
    }

    void main() {
      float pulse = sin(time * 8.0) * 0.5 + 0.5;
      float noise = hash(vUv * time);
      
      // Efeito de "flicker" quântico: se a prob é baixa, a aresta desaparece aleatoriamente
      if (noise > (probability + 0.2 * pulse)) {
        discard;

      float glow = 0.5 + 0.5 * sin(vUv.x * 10.0 - time * 5.0);
      vec3 finalColor = color * (1.0 + glow * (1.0 - probability));
      gl_FragColor = vec4(finalColor, probability * 0.8);
    }
  `
};

  }
```

### 2. O Fluxo de Pensamento (Kinetic Heat Flow)

Este motor renderiza partículas que viajam pelas geodésicas hiperbólicas, representando a ativação do Heat Kernel (Chebyshev).

**Arquivo:** `src/engine/FlowEngine.ts`

```typescript
import * as THREE from 'three';
import { calculateGeodesic } from '../math/poincare';

export class FlowEngine {
  private particles: THREE.InstancedMesh;
  private dummy = new THREE.Object3D();
  private particleData: {edgeId: string, progress: number, speed: number}[] = [];

  constructor(scene: THREE.Scene, count: number = 10000) {
    const geo = new THREE.SphereGeometry(0.004, 6, 6);
    const mat = new THREE.MeshBasicMaterial({ color: '#ff00ff', transparent: true, opacity: 0.9 });
    this.particles = new THREE.InstancedMesh(geo, mat, count);
    scene.add(this.particles);
  }

  // Injetar calor nas arestas ativas vindo do DIFFUSE do NietzscheDB
  injectHeat(activeEdges: any[]) {
    this.particleData = activeEdges.map(edge => ({
      edgeId: edge.id,
      geodesic: calculateGeodesic(edge.source, edge.target),
      progress: Math.random(),
      speed: 0.005 + (edge.activation * 0.02),
      activation: edge.activation
    }));
  }

  update(time: number) {
    this.particleData.forEach((p, i) => {
      p.progress = (p.progress + p.speed) % 1.0;
      const pos = this.getHyperbolicPoint(p.geodesic, p.progress);
      
      this.dummy.position.set(pos.x, pos.y, 0);
      this.dummy.scale.setScalar(p.activation * (1.0 + 0.5 * Math.sin(time * 10.0)));
      this.dummy.updateMatrix();
      this.particles.setMatrixAt(i, this.dummy.matrix);
    });
    this.particles.instanceMatrix.needsUpdate = true;
    this.particles.count = this.particleData.length;
  }

  private getHyperbolicPoint(geo: any, t: number) {
    const angle = geo.startAngle + (geo.endAngle - geo.startAngle) * t;
    return {
      x: geo.centerX + Math.cos(angle) * geo.radius,
      y: geo.centerY + Math.sin(angle) * geo.radius
    };
  }
}
```
